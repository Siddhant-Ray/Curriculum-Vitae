% Omit the introductory "In" supplied by BibTeX for conference venues.
\ExplSyntaxOn
\NewDocumentCommand{\publicationvenue}{m}{
  \tl_set:Nn \l_tmpa_tl {#1}
  \regex_replace_once:nnN {\A\s*In\s+} {} \l_tmpa_tl
  \tl_use:N \l_tmpa_tl
}
\ExplSyntaxOff

% Join separately marked first and last names before applying one underline.
\newcommand{\publicationauthors}[1]{%
  \begingroup
  \def\authorlist{#1}%
  \patchcmd{\authorlist}{\textbf{Siddhant} \textbf{Ray}}{\textbf{Siddhant Ray}}{}{}%
  \let\textbf\underline
  \authorlist
  \endgroup
}
% The shared style emits authors, title, then venue separated by newblock.
% Reorder those blocks for this layout without editing generated .bbl files.
\long\def\publicationblocks#1\newblock#2\newblock#3\publicationend{%
  {\bfseries #2}\par\nobreak
  \publicationauthors{#1}\par\nobreak
  \publicationvenue{#3}%
}
\newcommand{\publicationitem}[3][0pt]{%
  \par\addvspace{3pt}\needspace{4\baselineskip}%
  \entryheading{\small\publicationblocks#3\publicationend}{#2}%
  \addvspace{#1}%
}
